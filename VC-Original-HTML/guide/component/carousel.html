Carousel (SwiperJS)

마우스 스와이프, 부드러운 전환 효과, 완벽한 오토플레이 제어를 지원하는 프리미엄 SwiperJS 기반 캐러셀입니다.

1. Overview & Specs

ClassDescription
.swiper캐러셀 최상위 래퍼
.swiper-wrapper슬라이드들을 감싸는 래퍼
.swiper-slide개별 슬라이드 영역
.swiper-pagination하단 페이지네이션(점) 래퍼
.swiper-button-prev, .next내부 양옆 좌우 이동 버튼

  });
</script>

<!-- Swiper JS 초기화 옵션으로 여러 슬라이드 동시 노출 제어 -->
<script>
  new Swiper('.swiperMulti', {
    slidesPerView: 3,
    spaceBetween: 24,
    grabCursor: true,
    pagination: { el: '.swiper-pagination', clickable: true },
    breakpoints: {
      0: { slidesPerView: 1.2 },
      768: { slidesPerView: 2.2 },
      1024: { slidesPerView: 3 }
    }
  });
</script>

<!-- Swiper JS Autoplay API 활용 -->
<script>
  const swiperAuto = new Swiper('.swiperAutoplay', {
    loop: true,
    grabCursor: true,
    autoplay: {
      delay: 2500,
      disableOnInteraction: false, // 스와이프 후에도 오토플레이 유지
    }
  });

  const btnPlay = document.getElementById('swpPlay');
  const btnPause = document.getElementById('swpPause');

  btnPlay.addEventListener('click', () => {
    swiperAuto.autoplay.start();
    btnPlay.classList.add('active');
    btnPause.classList.remove('active');
  });

  btnPause.addEventListener('click', () => {
    swiperAuto.autoplay.stop();
    btnPause.classList.add('active');
    btnPlay.classList.remove('active');
  });
</script>
